## 3. 架构落地分析

### 3.1 改造形态:纯测试基建,src/ 零触碰

P0-3 的"架构"是 **在 `test/` 下加一层 GOLDEN 配置驱动的 byte-compare 套件**,不动 `src/**.ts` 一行,不动 `scripts/check-schema.mjs` 一行。核心数据流:

```
┌─────────────────────────────────────────────────────────────────────────┐
│  npm run render:golden  (开发者主动刷新)                                │
│    ↓                                                                    │
│  tsc -p tsconfig.test.json → dist-test/                                 │
│    ↓                                                                    │
│  node dist-test/scripts/render-golden.js                                │
│    ↓ import                                                             │
│  GOLDEN 数组(test/golden/golden.config.ts)                              │
│    ↓ 对每条 case                                                        │
│  render(tool, fixturePath) → {svg?,png?}  (test/golden/render.ts)       │
│    ↓ import from                                                        │
│  dist/{qr,formula,chart,card,render-svg,diagram/render}.js (现有产物)   │
│    ↓ writeFile                                                          │
│  test/golden/expected/<file>  (commit 入 git,作为 golden 真源)           │
└─────────────────────────────────────────────────────────────────────────┘

┌─────────────────────────────────────────────────────────────────────────┐
│  npm test / CI  (每次 push 自动跑)                                      │
│    ↓                                                                    │
│  tsc -p tsconfig.test.json → dist-test/                                 │
│    ↓                                                                    │

│    ↓                                                                    │
│  golden.test.ts:对 GOLDEN 每条 case                                     │

│    ↓ 读 test/golden/expected/<file>                                     │
│    ↓ byte 比对                                                          │
│  fail → "fresh render differs from ...; run npm run render:golden"      │
│    ↓                                                                    │
│  determinism.test.ts:grep 6 文件无 time/random + formula/chart 2× 自检  │
│    ↓                                                                    │
│  node scripts/check-schema.mjs (G1/G2/G3,不动)                          │
└─────────────────────────────────────────────────────────────────────────┘
```

**关键设计点**:

1. **GOLDEN 数组是单一真相源**:`test/golden/golden.config.ts` 一个文件,刷新器(`scripts/render-golden.mjs`)和验证器(`test/golden.test.ts`)都从这里读 case 列表。Archify 范式 1:1 搬。任何新增工具的 golden 用例只能在 GOLDEN 数组追加,不允许测试文件里另起。
2. **render() 分派抽到共享文件**:`test/golden/render.ts`(PRD §B3 open_point 14 已点名)。这样刷新器(写盘)和验证器(比对)走同一段分派逻辑,保证不会出现"刷新时用一套参数、验证时用另一套"的漂移。
3. **6 工具分派**(在 `render.ts` 内):
   - `qrcode`:`.txt` → `{text, format:"svg"}`;`.json` → 完整参数(JSON 直接 spread)
   - `formula`:`.tex` → `{tex, format:"svg"}`
   - `chart`:`.json` → `{spec, format:"svg"}`(JSON 是 Vega-Lite spec)
   - `card`:`.json` → `{...props, format:"svg"}`
   - `render_svg`:`.svg` → `{svg, format:"svg"}`(passthrough)
   - `diagram`:`.d2` → `{code, engine:"d2", format:"svg"}`;`.dot` → `{code, engine:"graphviz", format:"svg"}`
4. **比对策略 3 档**:`svg-byte`(SVG 全文 ===,formula 需 normalizeMathJaxIds)| `png-byte`(stripPngMetadata 后 Buffer.equals)| `qr-png-verify`(byte + jsQR 双校验)
5. **PRD §4.4 helpers.ts 需加第 6 个 helper**:`normalizeMathJaxIds(svg)` —— `s.replace(/MJX-\d+-/g, "MJX-N-")`(对非 MathJax SVG 是 no-op,可安全在所有 svg-byte 路径上调用,也可仅在 formula case 上调用)。**这是本阶段相对 PRD 的唯一接口增量**(R-01)。

#### 4.4.2 `test/golden/helpers.ts`(6+1 helpers)

```ts
import { readFileSync } from "node:fs";
import { PNG } from "pngjs";
import jsQR from "jsqr";

/** Archify 直搬:唯一规范化,只处理 CRLF。不过度规范化(会掩盖真回归)。 */
export function normalizeNewlines(text: string): string {
  return text.replace(/\r\n?/g, "\n");
}

/** R-01 新增:抹 MathJax SVG 的 MJX-N- 自增计数器(lazy singleton 内部状态)。
 *  对非 MathJax SVG 是 no-op(无 MJX- 前缀时 replace 不命中),可安全在所有 svg-byte 路径上调用。 */
export function normalizeMathJaxIds(svg: string): string {
  return svg.replace(/MJX-\d+-/g, "MJX-N-");
}

/** SVG byte-compare:规范化后全文 ===。formula 自动走 normalizeMathJaxIds(无副作用)。 */
export function compareSvg(fresh: string, checked: string, opts?: { preNormalize?: (s: string) => string }): { ok: boolean; diff?: string } {
  const a = normalizeNewlines(opts?.preNormalize ? opts.preNormalize(fresh) : normalizeMathJaxIds(fresh));
  const b = normalizeNewlines(opts?.preNormalize ? opts.preNormalize(checked) : normalizeMathJaxIds(checked));
  if (a === b) return { ok: true };
  const i = [...a].findIndex((c, idx) => c !== b[idx]);
  return {
    ok: false,
    diff: `first diff at char ${i}: fresh=${JSON.stringify(a.slice(Math.max(0, i - 40), i + 40))} checked=${JSON.stringify(b.slice(Math.max(0, i - 40), i + 40))}`,
  };
}

/** PNG 元数据 strip:丢弃 tEXt/zTXt/iTXt/tIME/eXIf/pHYs,只保留 IHDR+PLTE/tRNS+IDAT+IEND。
 *  等价 pngcrush -ow -rem allb。规避 resvg/libpng 不同版本写不同 Software 字段。 */
export function stripPngMetadata(buf: Buffer): Buffer {
  const KEEP = new Set(["IHDR", "PLTE", "tRNS", "IDAT", "IEND"]);
  const out: Buffer[] = [buf.subarray(0, 8)]; // signature
  let off = 8;
  while (off < buf.length) {
    const len = buf.readUInt32BE(off);
    const type = buf.toString("ascii", off + 4, off + 8);
    const chunkEnd = off + 12 + len; // 4(len)+4(type)+len(data)+4(CRC)
    if (KEEP.has(type)) out.push(buf.subarray(off, chunkEnd));
    off = chunkEnd;
  }
  return Buffer.concat(out);
}

/** PNG byte-compare:strip 后 Buffer.equals()。 */
export function comparePng(fresh: Buffer, checked: Buffer): { ok: boolean; reason?: string } {
  const a = stripPngMetadata(fresh);
  const b = stripPngMetadata(checked);
  if (a.equals(b)) return { ok: true };
  return { ok: false, reason: `fresh ${fresh.length}B / checked ${checked.length}B; after strip: ${a.length}B vs ${b.length}B` };
}

/** QR 双校验:byte-compare + jsQR 解码回原文等比。 */
export function verifyQrPng(png: Buffer, expectedText: string): { ok: boolean; reasons: string[] } {
  const reasons: string[] = [];
  const parsed = PNG.sync.read(png);
  const decoded = jsQR(new Uint8ClampedArray(parsed.data), parsed.width, parsed.height);
  if (!decoded) reasons.push("jsQR 解码失败(扫码器可能识别不出)");
  else if (decoded.data !== expectedText) reasons.push(`jsQR 解码 ≠ 原文:得 ${JSON.stringify(decoded.data)}`);
  return { ok: reasons.length === 0, reasons };
}

/** Archify NaN 守门:先决保证,任何 NaN/undefined 都不会污染 golden。 */
export function assertNoNaNOrUndefined(svg: string): void {
  if (/NaN|undefined/.test(svg)) {
    throw new Error(`rendered SVG contains NaN/undefined (would corrupt golden): ${svg.slice(0, 200)}...`);
  }
}
```

```ts
import { getDiagramEngine } from "../../dist/diagram/render.js";

const FIXTURES = path.resolve("test/golden/fixtures");

/** 渲染分派:fixturePath → {svg?, png?, input}(input 给 qr-png-verify 用)。 */
export async function render(tool: string, fixturePath: string): Promise<{ svg?: string; png?: Buffer; input: any }> {
  const abs = path.join(FIXTURES, fixturePath);
  if (tool === "qrcode") {
    if (fixturePath.endsWith(".txt")) {
      const text = readFileSync(abs, "utf8").trim();
      return { input: text, ...(await renderQR({ text, format: "svg" })) };
    }
    const params = JSON.parse(readFileSync(abs, "utf8"));
    return { input: params.text, ...(await renderQR(params)) };
  }
  if (tool === "formula") {
    const tex = readFileSync(abs, "utf8").trim();
    return { input: tex, ...(await renderFormula({ tex, format: "svg" })) };
  }
  if (tool === "chart") {
    const spec = JSON.parse(readFileSync(abs, "utf8"));
    return { input: spec, ...(await renderChart({ spec, format: "svg" })) };
  }
  if (tool === "card") {
    const props = JSON.parse(readFileSync(abs, "utf8"));
    return { input: props, ...(await renderCard({ ...props, format: "svg" })) };
  }
  if (tool === "render_svg") {
    const svg = readFileSync(abs, "utf8");
    return { input: svg, ...(await renderSvg({ svg, format: "svg" })) };
  }
  if (tool === "diagram") {
    const code = readFileSync(abs, "utf8");
    const engineName = fixturePath.endsWith(".dot") ? "graphviz" : "d2";
    const engine = getDiagramEngine(engineName)!;
    return { input: code, ...(await engine.render({ code, engine: engineName as any, format: "svg" })) };
  }
  throw new Error(`unknown tool: ${tool}`);
}
```
